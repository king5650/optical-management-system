🔧 Core Logic for Sending Email (Python Example)
import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText

def send_email(sender, password, recipients, subject, message):
    msg = MIMEMultipart()
    msg['From'] = sender
    msg['To'] = ", ".join(recipients)
    msg['Subject'] = subject
    msg.attach(MIMEText(message, 'plain'))

    server = smtplib.SMTP('smtp.gmail.com', 587)
    server.starttls()
    server.login(sender, password)
    server.sendmail(sender, recipients, msg.as_string())
    server.quit()

from PySide6.QtWidgets import (
    QDialog, QLabel, QLineEdit, QTextEdit, QPushButton,
    QListWidget, QListWidgetItem, QFileDialog, QVBoxLayout,
    QHBoxLayout, QMessageBox

from PySide6.QtCore import Qt, Signal


class SendMailDialog(QDialog):
    mailSent = Signal(dict)

    def __init__(self, people_list: list, parent=None):
        """
        people_list: list of dicts -> {"id","name","email"}
        """
        super().__init__(parent)
        self.setWindowTitle("Send Mail")
        self.setMinimumSize(580, 480)

        self.people_list = people_list
        self.attachments = []

        self.apply_styles()
        self.build_ui()

    # ------------------------------------------------------------
    # UI BUILDING
    # ------------------------------------------------------------
    def build_ui(self):
        layout = QVBoxLayout()

        title = QLabel("Compose Mail")
        title.setStyleSheet("font-size: 22px; font-weight: bold; color: #1565C0;")
        layout.addWidget(title)

        # RECIPIENT SELECTOR
        layout.addWidget(QLabel("Select Recipients:"))

        self.recipient_list = QListWidget()
        self.recipient_list.setSelectionMode(QListWidget.MultiSelection)

        # FIXED: display name but store email
        for person in self.people_list:
            text = f"{person['name']}  <{person['email']}>"
            item = QListWidgetItem(text)
            item.setData(Qt.UserRole, person["email"])
            self.recipient_list.addItem(item)

        layout.addWidget(self.recipient_list)

        # SUBJECT
        layout.addWidget(QLabel("Subject:"))
        self.subject_input = QLineEdit()
        layout.addWidget(self.subject_input)

        # MESSAGE
        layout.addWidget(QLabel("Message:"))
        self.message_input = QTextEdit()
        layout.addWidget(self.message_input)

        # ATTACHMENTS
        attach_btn = QPushButton("Attach File")
        attach_btn.clicked.connect(self.pick_attachment)
        layout.addWidget(attach_btn)

        # BUTTONS
        btns = QHBoxLayout()
        cancel = QPushButton("Cancel")
        cancel.clicked.connect(self.reject)

        send = QPushButton("Send Mail")
        send.clicked.connect(self.send_mail)

        btns.addWidget(cancel)
        btns.addWidget(send)

        layout.addLayout(btns)

        self.setLayout(layout)

    # ------------------------------------------------------------
    # FILE PICKER
    # ------------------------------------------------------------
    def pick_attachment(self):
        file, _ = QFileDialog.getOpenFileName(self, "Select File")
        if file:
            self.attachments.append(file)
            QMessageBox.information(self, "Attachment Added", f"Added: {file}")

    # ------------------------------------------------------------
    # SEND MAIL
    # ------------------------------------------------------------
    def send_mail(self):
        selected_items = self.recipient_list.selectedItems()

        if not selected_items:
            QMessageBox.warning(self, "No Recipients", "Select at least one recipient.")
            return

        # FIXED: Extract email addresses only
        recipients = [item.data(Qt.UserRole) for item in selected_items]

        subject = self.subject_input.text().strip()
        message = self.message_input.toPlainText().strip()

        if not subject or not message:
            QMessageBox.warning(self, "Empty Fields", "Subject and message are required.")
            return

        mail_data = {
            "recipients": recipients,
            "subject": subject,
            "message": message,
            "attachments": self.attachments
        }

        self.mailSent.emit(mail_data)
        QMessageBox.information(self, "Success", "Mail queued for sending.")
        self.accept()
